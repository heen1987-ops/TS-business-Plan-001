        <script>
        (function () {
    const carousel = document.getElementById('carousel-images');
    const visibleImages = 3; // Number of images visible at once
    const interval = 5000; // Time between transitions in ms
    let currentIndex = 0;

    // Clone the first few images and append them to the end
    const images = Array.from(carousel.children);
    for (let i = 0; i < visibleImages; i++) {
        const clone = images[i].cloneNode(true);
        carousel.appendChild(clone);
    }

    // Set the width of the carousel dynamically based on visible images
    const imageWidthPercentage = 100 / visibleImages;
    carousel.style.display = 'flex';
    carousel.style.width = `${images.length + visibleImages}00%`; // Account for clones
    Array.from(carousel.children).forEach((img) => {
        img.style.flex = `0 0 ${imageWidthPercentage}%`; // Ensure each image takes only its share
    });

    function slide() {
        currentIndex++;
        carousel.style.transition = 'transform 0.5s ease'; // Smooth transition
        carousel.style.transform = `translateX(-${currentIndex * imageWidthPercentage}%)`;

        // Reset to the start seamlessly when reaching the end of the loop
        if (currentIndex >= images.length) {
            setTimeout(() => {
                carousel.style.transition = 'none'; // Remove transition for seamless reset
                carousel.style.transform = 'translateX(0)';
                currentIndex = 0;
            }, 500); // Match the transition duration
        }
    }

    setInterval(slide, interval);
})();

    </script>
