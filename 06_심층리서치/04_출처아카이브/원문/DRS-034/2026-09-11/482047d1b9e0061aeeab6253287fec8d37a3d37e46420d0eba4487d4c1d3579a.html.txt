	function goBack() {		
		var preUrl = document.referrer;                      // 전 페이지 url
		var iniUrl = sessionStorage.getItem('iniUrl');	     // 최초 팝업창 url

		if(window.history.length > 1){                       // 최초 팝업창(length=1)이 아니라면
			if(iniUrl == preUrl) {                           // 뒤로갈 페이지가 처음 열린 팝업창일 때
				sessionStorage.setItem('equalUrl', 'init');				
			} else {
				sessionStorage.setItem('equalUrl', 'diff');
			}		
			window.history.back();                           // 뒤로 가기			
		} 
	}
	
	/* 뒤로가기 버튼 세팅 */
	function updateBackBtn() {
		var backBtn = document.getElementById('backBtn');
 				
		if (window.history.length == 1) {                    // 처음 팝업을 열었을때 길이는 1
			backBtn.style.display = 'none'; 			
			// 최초의 열린 팝업 url 저장
			sessionStorage.setItem('iniUrl', window.location.href);

		} else {                                             // history 길이가 2 이상  
			var equalUrl = sessionStorage.getItem('equalUrl');
			
			if (equalUrl == "init"){
				backBtn.style.display = 'none';              // 최초 url과 동일할 경우 뒤로가기 아이콘 숨김
				sessionStorage.setItem('equalUrl', 'diff');		
						
			} else {
				backBtn.style.display = 'inline-block';      // diff 인  경우
			}
		} 
	}
